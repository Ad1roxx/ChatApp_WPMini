# The AI service. Kept deliberately small: a web framework, a driver, and the
# embedding model. No LangChain or LlamaIndex — the whole point of this service
# is that the retrieval logic is visible and testable rather than configured.
fastapi>=0.115,<1

pymongo>=4.9,<5
python-dotenv>=1,<2

# --- S1: the embedding stack, pinned exactly ---
# Everything that decides what a vector *is* is pinned, because a change here
# does not fail — it silently produces different vectors for the same text.
# Half the index would then answer queries with the old geometry and half with
# the new one, and nothing about the index would look wrong. The version a
# measurement was taken with is part of the measurement.
sentence-transformers==6.1.0
transformers==5.17.0
torch==2.14.0
numpy==2.4.6

# --- S1: document extraction ---
# Pinned for the same reason: a change in how text comes out of a PDF changes
# every chunk derived from it, and the index would still look valid afterwards.
PyMuPDF==1.28.2
python-docx==1.2.0

# --- S2: generation ---
# The two providers are called over plain HTTP rather than through their SDKs,
# which retry 429s on their own policy and hide the rate-limit headers this
# service enforces (providers.py says why in full). Pinned because the
# retry and timeout behaviour the tests prove is httpx's.
httpx==0.28.1
